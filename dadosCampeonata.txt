import Data.List (nub, sort, foldl, sortOn)

-- camada 3.1 tipos

type Time = String

data Resultado = Vitoria | Empate | Derrota   -- data cria um tipo novo
  deriving (Show, Eq)                         -- Show: print funciona; Eq: == funciona

data Partida = Partida
  { rodada        :: Int
  , mandante      :: Time
  , golsMandante  :: Int
  , golsVisitante :: Int
  , visitante     :: Time
  } deriving (Show, Eq)

data Linha = Linha
  { time       :: Time
  , jogos      :: Int
  , vitorias   :: Int
  , empates    :: Int
  , derrotas   :: Int
  , golsPro    :: Int
  , golsContra :: Int
  } deriving (Show, Eq)


participou :: Time -> Partida -> Bool
participou t part = t == mandante part || t == visitante part

resultadoDoPlacar :: Int -> Int -> Resultado
resultadoDoPlacar meus deles
  | meus > deles = Vitoria
  | meus < deles = Derrota
  | otherwise    = Empate


resultadoPara :: Time -> Partida -> Maybe Resultado
resultadoPara t part
  | t == mandante part  = Just (resultadoDoPlacar (golsMandante part) (golsVisitante part))
  | t == visitante part = Just (resultadoDoPlacar (golsVisitante part) (golsMandante part))
  | otherwise           = Nothing


times :: [Partida] -> [Time]
times ps = sort (nub (map mandante ps ++ map visitante ps))

-- Camada 3.2
pontos :: Resultado -> Int
pontos Vitoria  = 3
pontos Empate   = 1
pontos Derrota  = 0

ehVitoria :: Resultado -> Int
ehVitoria Vitoria = 1
ehVitoria _       = 0

ehEmpate :: Resultado -> Int
ehEmpate Empate = 1
ehEmpate _      = 0

ehDerrota  :: Resultado -> Int
ehDerrota Derrota = 1
ehDerrota _       = 0

linhaDoTime :: [Partida] -> Time -> Linha
linhaDoTime ps t = foldl atualizar linhaVazia (filter (participou t) ps)
  where
    linhaVazia = Linha t 0 0 0 0 0 0

    atualizar linha partida =
      let (meusGols, golsAdversario)
            | mandante partida == t = (golsMandante partida, golsVisitante partida)
            | otherwise             = (golsVisitante partida, golsMandante partida)
          resultado = resultadoDoPlacar meusGols golsAdversario
      in linha
           { jogos      = jogos linha + 1
           , vitorias   = vitorias linha + ehVitoria resultado
           , empates    = empates linha  + ehEmpate resultado
           , derrotas   = derrotas linha + ehDerrota resultado
           , golsPro    = golsPro linha    + meusGols
           , golsContra = golsContra linha + golsAdversario
           }

pontosLinha :: Linha -> Int
pontosLinha l = vitorias l * pontos Vitoria + empates l * pontos Empate

saldo :: Linha -> Int
saldo l = golsPro l - golsContra l

aproveitamento :: Linha -> Double
aproveitamento l
      | jogos l == 0 = 0
      | otherwise    = fromIntegral (round (valor * 100)) / 100
      where
        valor = fromIntegral (pontosLinha l) / fromIntegral (jogos l * 3) * 100
        
classificacao :: [Partida] -> [Linha]
classificacao ps = reverse (sortOn chave linhas)
  where
    linhas = map (linhaDoTime ps) (times ps)
    chave l = (pontosLinha l, vitorias l, saldo l, golsPro l)
    
ateRodada :: Int -> [Partida] -> [Partida] 
ateRodada n ps = [p | p <- ps, rodada p <= n]
