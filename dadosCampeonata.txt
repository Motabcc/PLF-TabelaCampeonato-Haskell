import Data.List (nub, sort)

-- camada 3.1 tipos

type Time = String

data Resultado = Vitoria | Empate | Derrota   -- data cria um tipo novo
  deriving (Show, Eq)                         -- Show: print funciona; Eq: == funciona

data Partida = Partida
  { rodada        :: Int
  , mandante      :: Time
  , golsMandante  :: Int
  , golsVisitante :: Int
  , visitante     :: Time
  } deriving (Show, Eq)

data Linha = Linha
  { time       :: Time
  , jogos      :: Int
  , vitorias   :: Int
  , empates    :: Int
  , derrotas   :: Int
  , golsPro    :: Int
  , golsContra :: Int
  } deriving (Show, Eq)


participou :: Time -> Partida -> Boole
participou t part = t == mandante part || t == visitante part

resultadoDoPlacar :: Int -> Int -> Resultado
resultadoDoPlacar meus deles
  | meus > deles = Vitoria
  | meus < deles = Derrota
  | otherwise    = Empate


resultadoPara :: Time -> Partida -> Maybe Resultado
resultadoPara t part
  | t == mandante part  = Just (resultadoDoPlacar (golsMandante part) (golsVisitante part))
  | t == visitante part = Just (resultadoDoPlacar (golsVisitante part) (golsMandante part))
  | otherwise           = Nothing


times :: [Partida] -> [Time]
times ps = sort (nub (map mandante ps ++ map visitante ps))
